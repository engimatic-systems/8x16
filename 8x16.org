* Package =ui8x16=

The package boundary is intentionally small in PR 1: public value types only,
with no drawing backend dependency.

The frame pipeline separates interaction from lowering. Interaction queries
consume sampled input and update caller-owned application state; lowering turns
that state into primitive draw commands.

#+begin_example
sample platform input
  -> ui_begin_frame
  -> interaction/query phase
  -> caller updates app state
  -> lowering phase
  -> ordered primitive draw commands
  -> backend render
#+end_example

package ui8x16
#+end_src

	pressed_id: [MOUSE_BUTTON_COUNT]UI_ID,

	hit_any:       bool,
	activated_any: bool,

#+end_src

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name mouse-button-index
mouse_button_index :: proc(button: Mouse_Button) -> int {
	return int(button)
}
#+end_src

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name ui-begin-frame
ui_begin_frame :: proc(state: ^UI_State, input: Mouse_Input) {
	for i in 0..<MOUSE_BUTTON_COUNT {
		if state.previous_down[i] && !state.mouse.down[i] {
			state.pressed_id[i] = UI_ID_NONE
		}

		state.previous_down[i] = state.mouse.down[i]
	}

	state.mouse = input
	state.hit_any = false
	state.activated_any = false
}
#+end_src

** Activation

Interaction queries are separate from command lowering. A typical frame samples
platform input, calls =ui_begin_frame=, queries interaction state with these
procedures, lets the caller update application state, then lowers semantic UI
surfaces into draw commands.

=ui_hit= marks advisory =hit_any=. =ui_activate= records press origin and
returns true only when press and release both occur inside the same =id, rect=
pair. =ui_pressed= reports whether a recorded press is currently held for an
=id=; callers pass that value into button/toggle lowering when they want pressed
visuals.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name ui-hit
ui_hit :: proc(state: ^UI_State, rect: Cell_Rect) -> bool {
	hit := contains(rect, state.mouse.x, state.mouse.y)
	if hit {
		state.hit_any = true
	}
	return hit
}
#+end_src

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name ui-activate
ui_activate :: proc(state: ^UI_State, id: UI_ID, rect: Cell_Rect, button: Mouse_Button = .Left) -> bool {
	index := mouse_button_index(button)
	hit := ui_hit(state, rect)
	pressed := !state.previous_down[index] && state.mouse.down[index]
	released := state.previous_down[index] && !state.mouse.down[index]

	if pressed && hit {
		state.pressed_id[index] = id
	}

	activated := released &&
	             hit &&
	             state.pressed_id[index] == id
	if activated {
		state.activated_any = true
	}
	return activated
}
#+end_src

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name ui-pressed
ui_pressed :: proc(state: ^UI_State, id: UI_ID, button: Mouse_Button = .Left) -> bool {
	index := mouse_button_index(button)
	return state.mouse.down[index] && state.pressed_id[index] == id
}
#+end_src

** Geometry Constructors

Constructors assert positive dimensions. Callers should not pass invalid
rectangles; checked builds make that failure immediate.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name cell-rect-proc
cell_rect :: proc(col, row, w, h: int) -> Cell_Rect {
	assert(w > 0, "Cell_Rect width must be positive")
	assert(h > 0, "Cell_Rect height must be positive")
	return Cell_Rect{col = col, row = row, w = w, h = h}
}
#+end_src

=rect_rel= composes coordinates without retaining parent/child relationships.
The returned rectangle is absolute.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name rect-rel
rect_rel :: proc(parent: Cell_Rect, col, row, w, h: int) -> Cell_Rect {
	return cell_rect(parent.col + col, parent.row + row, w, h)
}
#+end_src

** Pixel Conversion

=rect_px= translates cell-space geometry to pixels using the fixed 8x16 metric.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name rect-px
rect_px :: proc(rect: Cell_Rect) -> Pixel_Rect {
	return Pixel_Rect{
		x = rect.col * CELL_WIDTH,
		y = rect.row * CELL_HEIGHT,
		w = rect.w * CELL_WIDTH,
		h = rect.h * CELL_HEIGHT,
	}
}
#+end_src

** Hit Testing

Hit testing is half-open in pixel space: left/top edges are inside, right/bottom
edges are outside.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name contains
contains :: proc(rect: Cell_Rect, x, y: int) -> bool {
	px := rect_px(rect)
	return x >= px.x &&
	       y >= px.y &&
	       x < px.x + px.w &&
	       y < px.y + px.h
}
#+end_src

** Primitive Draw Commands

The draw buffer is the handoff point between UI compilation and rendering. It
contains backend-ready primitive commands only: filled rectangles, stroked
rectangles, and text draws. Semantic UI concepts compile to these commands
before a renderer sees them.

Each primitive has its own payload type. This increases the public type surface,
but it names the renderable data directly and keeps each payload narrow.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name fill-rect-command
Fill_Rect_Command :: struct {
	rect:  Pixel_Rect,
	color: Color,
}
#+end_src

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name stroke-rect-command
Stroke_Rect_Command :: struct {
	rect:  Pixel_Rect,
	color: Color,
}
#+end_src

=Text_Command= carries a pixel origin rather than a rectangle. Text bounds and
overflow checks belong to layout/debug code before command emission; the backend
receives only the position, bytes, resolved color, and integer scale needed to
draw.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name text-command
Text_Command :: struct {
	pos:   Pixel_Point,
	text:  string,
	color: Color,
	scale: int,
}
#+end_src

=Draw_Command= is an Odin union over the primitive payloads. The tradeoff is
intentional: command storage may be larger and callers see more public types,
but the active-payload invariant is managed by Odin's type system instead of a
manual =kind= field plus inactive struct fields.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name draw-command
Draw_Command :: union {
	Fill_Rect_Command,
	Stroke_Rect_Command,
	Text_Command,
}
#+end_src

The buffer uses caller-owned storage. Commands are append-only and render in
buffer order; there is no z coordinate or hidden sorting step.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name draw-buffer
Draw_Buffer :: struct {
	commands: []Draw_Command,
	count:    int,
}
#+end_src

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name draw-buffer-proc
draw_buffer :: proc(storage: []Draw_Command) -> Draw_Buffer {
	return Draw_Buffer{commands = storage}
}
#+end_src

=text_scale_valid= constrains the fixed 8x16 design kernel to integer scales
=1|2|3|4=. Backends may choose how to realize those scales, but commands should
not carry arbitrary floating-point text sizes.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name text-scale-valid
text_scale_valid :: proc(scale: int) -> bool {
	return scale >= 1 && scale <= 4
}
#+end_src

=draw_push= is the only raw append path. Overflow asserts loudly because silent
command loss would make UI failures hard to diagnose.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name draw-push
draw_push :: proc(buffer: ^Draw_Buffer, command: Draw_Command) {
	assert(buffer.count < len(buffer.commands), "Draw_Buffer overflow")
	buffer.commands[buffer.count] = command
	buffer.count += 1
}
#+end_src

The =draw_*= procedures are typed constructors for command emission. They build
the correct union variant, append it, and keep call sites from spelling out the
union conversion directly.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name draw-fill-rect
draw_fill_rect :: proc(buffer: ^Draw_Buffer, rect: Pixel_Rect, color: Color) {
	draw_push(buffer, Draw_Command(Fill_Rect_Command{rect = rect, color = color}))
}
#+end_src

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name draw-stroke-rect
draw_stroke_rect :: proc(buffer: ^Draw_Buffer, rect: Pixel_Rect, color: Color) {
	draw_push(buffer, Draw_Command(Stroke_Rect_Command{rect = rect, color = color}))
}
#+end_src

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name draw-text
draw_text :: proc(buffer: ^Draw_Buffer, pos: Pixel_Point, text: string, color: Color, scale: int) {
	assert(text_scale_valid(scale), "Text scale must be 1..4")
	draw_push(buffer, Draw_Command(Text_Command{pos = pos, text = text, color = color, scale = scale}))
}
#+end_src

=ui_end= returns the ordered command slice for a renderer or test snapshot. It
does not clear storage; caller-owned frame lifetime remains explicit.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name ui-end
ui_end :: proc(buffer: ^Draw_Buffer) -> []Draw_Command {
	return buffer.commands[:buffer.count]
}
#+end_src

** Command Compiler

The compiler layer accepts UI semantics, resolves style to raw colors, performs
layout/debug checks, and emits primitive draw commands.

In this package, "lowering" means moving down the abstraction ladder: caller
intent becomes resolved colors, pixel geometry, debug warnings, and finally
ordered primitive commands. Interaction decisions happen immediately before this
lowering step, not inside it. The backend should only see the final primitive
command stream.

=UI_Style= is the style-resolution input. Semantic controls choose from these
colors before emission so the backend never needs to know about button, panel,
toggle, or bar states.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name ui-style
UI_Style :: struct {
	text: Color,
	line: Color,
	panel: Color,

	button_fill:        Color,
	button_text:        Color,
	button_active_fill: Color,
	button_active_text: Color,

	bar_fill:  Color,
	bar_empty: Color,
}
#+end_src

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name default-style
default_style :: proc() -> UI_Style {
	return UI_Style{
		text = Color{r = 230, g = 230, b = 230, a = 255},
		line = Color{r = 80, g = 80, b = 80, a = 255},
		panel = Color{r = 24, g = 24, b = 24, a = 255},

		button_fill = Color{r = 48, g = 48, b = 48, a = 255},
		button_text = Color{r = 235, g = 235, b = 235, a = 255},
		button_active_fill = Color{r = 220, g = 220, b = 220, a = 255},
		button_active_text = Color{r = 16, g = 16, b = 16, a = 255},

		bar_fill = Color{r = 96, g = 180, b = 120, a = 255},
		bar_empty = Color{r = 44, g = 44, b = 44, a = 255},
	}
}

=Debug_Buffer= collects compiler/debug warnings. These checks happen before
primitive emission because primitive commands intentionally carry no semantic
layout policy, such as text bounds.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name debug-warning-kind
Debug_Warning_Kind :: enum {
	Text_Overflow,
}
#+end_src

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name debug-warning
Debug_Warning :: struct {
	kind: Debug_Warning_Kind,
	id:   UI_ID,
	rect: Cell_Rect,

	text:            string,
	measured_width:  int,
	available_width: int,
}
#+end_src

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name debug-buffer
Debug_Buffer :: struct {
	warnings: []Debug_Warning,
	count:    int,

	overflowed: bool,
}
#+end_src

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name debug-push
debug_push :: proc(buffer: ^Debug_Buffer, warning: Debug_Warning) -> bool {
	if buffer == nil {
		return false
	}
	if buffer.count >= len(buffer.warnings) {
		buffer.overflowed = true
		return false
	}

	buffer.warnings[buffer.count] = warning
	buffer.count += 1
	return true
}
#+end_src

=debug_text_overflow= de-duplicates by =UI_ID= within a frame. Repeated calls
for the same semantic element should not flood the warning buffer.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name debug-text-overflow
debug_text_overflow :: proc(buffer: ^Debug_Buffer, id: UI_ID, rect: Cell_Rect, text: string, measured_width, available_width: int) -> bool {
	if buffer == nil {
		return false
	}

	for i in 0..<buffer.count {
		warning := buffer.warnings[i]
		if warning.kind == .Text_Overflow && warning.id == id {
			return false
		}
	}

	return debug_push(buffer, Debug_Warning{
		kind = .Text_Overflow,
		id = id,
		rect = rect,
		text = text,
		measured_width = measured_width,
		available_width = available_width,
	})
}
#+end_src

=UI= is the per-frame compiler context. It groups output command storage,
optional debug storage, and resolved style so semantic calls do not need to
thread those arguments independently. It does not own interaction state. Text
scale is not frame state: text-producing calls carry it explicitly and default
to scale =1=.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name ui-context
UI :: struct {
	draw:  ^Draw_Buffer,
	debug: ^Debug_Buffer,
	style: UI_Style,
}
#+end_src

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name ui-context-proc
ui_context :: proc {
	ui_context_default,
	ui_context_with_style,
}

ui_context_default :: proc(draw: ^Draw_Buffer, debug: ^Debug_Buffer) -> UI {
	return ui_context_with_style(draw, debug, default_style())
}

ui_context_with_style :: proc(draw: ^Draw_Buffer, debug: ^Debug_Buffer, style: UI_Style) -> UI {
	return UI{draw = draw, debug = debug, style = style}
}
#+end_src

These helpers are the layout-lowering layer. They convert semantic values into
bounded integers and pixel-space positions before primitive emission.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name int-clamp
int_clamp :: proc(value, low, high: int) -> int {
	if value < low {
		return low
	}
	if value > high {
		return high
	}
	return value
}
#+end_src

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name text-width-px
text_width_px :: proc(text: string, scale: int) -> int {
	assert(text_scale_valid(scale), "Text scale must be 1..4")
	return len(text) * CELL_WIDTH * scale
}
#+end_src

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name pixel-origin
pixel_origin :: proc(rect: Cell_Rect) -> Pixel_Point {
	px := rect_px(rect)
	return Pixel_Point{x = px.x, y = px.y}
}
#+end_src

=compile_text= is the text-specific lowering point. It measures the text against
the semantic rectangle, records overflow in the debug path, then emits a
primitive =Text_Command= with only origin, bytes, color, and scale. Scale is
per text emission so labels in the same frame can render at different integer
scales.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name compile-text
compile_text :: proc(ui: ^UI, id: UI_ID, rect: Cell_Rect, pos: Pixel_Point, text: string, color: Color, scale: int) {
	measured_width := text_width_px(text, scale)
	available_width := rect.w * CELL_WIDTH
	if measured_width > available_width {
		debug_text_overflow(ui.debug, id, rect, text, measured_width, available_width)
	}

	draw_text(ui.draw, pos, text, color, scale)
}
#+end_src

=ui_text= is the smallest semantic surface: it lowers the rectangle origin to
pixels, performs text debug checks, and emits text.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name ui-text
ui_text :: proc(ui: ^UI, id: UI_ID, rect: Cell_Rect, text: string, color: Color, scale: int = 1) {
	compile_text(ui, id, rect, pixel_origin(rect), text, color, scale)
}
#+end_src

=ui_box= and =ui_panel= lower framed regions into fill/stroke rectangles.
=ui_panel= is style-specific; =ui_box= accepts raw colors.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name ui-box
ui_box :: proc(ui: ^UI, id: UI_ID, rect: Cell_Rect, fill, line: Color) {
	_ = id
	px := rect_px(rect)
	draw_fill_rect(ui.draw, px, fill)
	draw_stroke_rect(ui.draw, px, line)
}
#+end_src

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name ui-panel
ui_panel :: proc(ui: ^UI, id: UI_ID, rect: Cell_Rect) {
	ui_box(ui, id, rect, ui.style.panel, ui.style.line)
}
#+end_src

=ui_bar= lowers a scalar value into three primitives: empty fill, value fill,
and outline. Value clamping belongs here because the backend should not know the
bar's maximum or semantic range.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name ui-bar
ui_bar :: proc(ui: ^UI, id: UI_ID, rect: Cell_Rect, value, max_value: int, fill: Color) {
	_ = id
	px := rect_px(rect)
	fill_px := px
	fill_px.w = 0
	if max_value > 0 {
		fill_px.w = px.w * int_clamp(value, 0, max_value) / max_value
	}

	draw_fill_rect(ui.draw, px, ui.style.bar_empty)
	draw_fill_rect(ui.draw, fill_px, fill)
	draw_stroke_rect(ui.draw, px, ui.style.line)
}
#+end_src

=ui_button= lowers a button visual into fill, stroke, and text primitives.
Interaction is handled before this call; callers pass =pressed=true= when the
interaction/query phase says this button is currently held. The label is inset
by 2 pixels before text compilation. The label scale defaults to =1= but remains
a per-call value.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name ui-button
ui_button :: proc(ui: ^UI, id: UI_ID, rect: Cell_Rect, label: string, scale: int = 1, pressed: bool = false) {
	fill := ui.style.button_fill
	text_color := ui.style.button_text
	if pressed {
		fill = ui.style.button_active_fill
		text_color = ui.style.button_active_text
	}

	ui_box(ui, id, rect, fill, ui.style.line)
	text_rect := rect
	text_px := rect_px(text_rect)
	text_px.x += 2
	compile_text(ui, id, rect, Pixel_Point{x = text_px.x, y = text_px.y}, label, text_color, scale)
}
#+end_src

=ui_toggle= shares the button shape but takes active state from the caller. It
does not flip that state. The caller handles activation in the interaction/query
phase, updates application state, and passes the current =active= value here.
Like buttons, toggle label scale defaults to =1= and pressed visual state is
carried per call.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name ui-toggle
ui_toggle :: proc(ui: ^UI, id: UI_ID, rect: Cell_Rect, label: string, active: bool, scale: int = 1, pressed: bool = false) {
	fill := ui.style.button_fill
	text_color := ui.style.button_text
	if active || pressed {
		fill = ui.style.button_active_fill
		text_color = ui.style.button_active_text
	}

	ui_box(ui, id, rect, fill, ui.style.line)
	text_rect := rect
	text_px := rect_px(text_rect)
	text_px.x += 2
	compile_text(ui, id, rect, Pixel_Point{x = text_px.x, y = text_px.y}, label, text_color, scale)
}
#+end_src

* Raylib Backend

The raylib backend is a separate package. It renders ordered primitive commands
and owns only backend concerns: font handles, texture filtering, and concrete
raylib draw calls.

The caller owns the raylib application context. This package assumes the window
has already been initialized and that rendering happens inside an active
=rl.BeginDrawing= / =rl.EndDrawing= pair. Callers remain responsible for
=rl.InitWindow=, =rl.SetTargetFPS=, =rl.BeginDrawing=, =rl.EndDrawing=, and
=rl.CloseWindow=.

This boundary keeps the backend narrow. It does not own style resolution, input
routing, hover detection, activation, text overflow policy, clipping policy, or
application canvas rendering.

#+begin_src odin :tangle raylib_8x16/raylib_8x16.odin :noweb no :comments no :name raylib-package
package raylib_8x16
#+end_src

#+begin_src odin :tangle raylib_8x16/raylib_8x16.odin :noweb no :comments no :name raylib-imports
import "core:c"
import "core:testing"
import rl "vendor:raylib"
import ui "../ui8x16"
#+end_src

=Renderer= is the happy-path backend handle. It always owns the font loaded by
=load_renderer= and unloads it through =unload_renderer=. Borrowed/shared font
construction can be added later if a caller needs that granularity; v0 keeps
lifetime rules visible and singular.

The caller provides scratch rune storage. Text rendering converts each command's
string into that caller-owned buffer before calling raylib, and fails loudly if
the buffer is too small.

#+begin_src odin :tangle raylib_8x16/raylib_8x16.odin :noweb no :comments no :name raylib-renderer
Renderer :: struct {
	font:       rl.Font,
	text_runes: []rune,
	spacing:    f32,
}
#+end_src

Conversion helpers translate core value types into raylib value types. They do

=load_renderer= loads the font and forces point filtering so the fixed 8x16
design kernel stays crisp at integer scales. =unload_renderer= releases the
owned font; callers should pair these calls around the raylib application
lifetime.

#+begin_src odin :tangle raylib_8x16/raylib_8x16.odin :noweb no :comments no :name raylib-renderer-procs
load_renderer :: proc(font_path: cstring, text_runes: []rune, spacing: f32 = 0) -> Renderer {
	font := rl.LoadFont(font_path)
	rl.SetTextureFilter(font.texture, .POINT)
	return Renderer{font = font, text_runes = text_runes, spacing = spacing}
}

unload_renderer :: proc(renderer: ^Renderer) {
	rl.UnloadFont(renderer.font)
}
#+end_src

=text_runes= is transient conversion scratch for the immediate draw call. The
returned slice aliases =renderer.text_runes= and is not stable after the next
text command.

#+begin_src odin :tangle raylib_8x16/raylib_8x16.odin :noweb no :comments no :name raylib-text-runes
text_runes :: proc(renderer: ^Renderer, text: string) -> []rune {
	assert(len(text) <= len(renderer.text_runes), "Renderer text rune scratch too small")
	for i in 0..<len(text) {
		renderer.text_runes[i] = rune(text[i])
	}
	return renderer.text_runes[:len(text)]
}
#+end_src

=render_command= is the only backend dispatch point. It switches over primitive
draw command payloads and issues concrete raylib calls in place. Text scale is
limited by the core =Text_Command= contract; raylib receives the resolved pixel
font size.

#+begin_src odin :tangle raylib_8x16/raylib_8x16.odin :noweb no :comments no :name raylib-render-command
render_command :: proc(renderer: ^Renderer, command: ui.Draw_Command) {
	switch payload in command {
	case ui.Fill_Rect_Command:
		rl.DrawRectangleRec(raylib_rect(payload.rect), raylib_color(payload.color))
	case ui.Stroke_Rect_Command:
		rl.DrawRectangleLinesEx(raylib_rect(payload.rect), 1, raylib_color(payload.color))
	case ui.Text_Command:
		runes := text_runes(renderer, payload.text)
		rl.DrawTextCodepoints(
			renderer.font,
			raw_data(runes),
			c.int(len(runes)),
			raylib_point(payload.pos),
			f32(ui.CELL_HEIGHT * payload.scale),
			renderer.spacing,
			raylib_color(payload.color),
		)
	}
}
#+end_src

=render_commands= preserves command-buffer order. There is no z coordinate and
no backend sorting.

#+begin_src odin :tangle raylib_8x16/raylib_8x16.odin :noweb no :comments no :name raylib-render-commands
render_commands :: proc(renderer: ^Renderer, commands: []ui.Draw_Command) {
	for command in commands {
		render_command(renderer, command)
	}
}
#+end_src

Tests assert command shape; visual smoke belongs to callers or examples that
create a raylib context.

* Runnable Raylib Examples

Runnable examples own the platform shell: raylib window setup, frame timing,
mouse sampling, direct canvas drawing, UI command compilation, and backend
rendering. They are intentionally outside the core packages so application
lifecycle code does not leak into =ui8x16= or =raylib_8x16=.

The drawing app is stateful. The UI controls update caller-owned app state, the
canvas is drawn directly with raylib as arbitrary application graphics, and the
8x16 UI is rendered over the same frame through the primitive command backend.
It uses the bundled Oldschool PC Font Pack VGA font in =assets/fonts=.

Run it from the repository root:

#+begin_src sh
odin run examples/drawing_app.odin -file
#+end_src

#+begin_src odin :tangle examples/drawing_app.odin :noweb no :comments no :name example-drawing-app
package main

import "core:c"
import rl "vendor:raylib"
import ui "../ui8x16"
import rui "../raylib_8x16"

WINDOW_WIDTH :: 400
WINDOW_HEIGHT :: 320

FONT_PATH :: "assets/fonts/PxPlus_IBM_VGA_8x16.ttf"

CANVAS_X :: 176
CANVAS_Y :: 64
CANVAS_COLS :: 64
CANVAS_ROWS :: 64
PIXEL_SIZE :: 3

MAX_BRUSH_SIZE :: 8

ID_CLEAR :: ui.UI_ID(402)
ID_ERASE :: ui.UI_ID(403)
ID_GRID  :: ui.UI_ID(404)
ID_BRUSH_DOWN :: ui.UI_ID(407)
ID_BRUSH_UP   :: ui.UI_ID(408)

Canvas :: struct {
	cells: [CANVAS_COLS * CANVAS_ROWS]bool,
}

App_State :: struct {
	erase:      bool,
	show_grid:  bool,
	brush_size: int,
}

raylib_mouse_input :: proc() -> ui.Mouse_Input {
	return ui.Mouse_Input{
		x = int(rl.GetMouseX()),
		y = int(rl.GetMouseY()),
		down = [ui.MOUSE_BUTTON_COUNT]bool{
			rl.IsMouseButtonDown(.LEFT),
			rl.IsMouseButtonDown(.MIDDLE),
			rl.IsMouseButtonDown(.RIGHT),
		},
	}
}

canvas_index :: proc(col, row: int) -> int {
	return row * CANVAS_COLS + col
}

clear_canvas :: proc(canvas: ^Canvas) {
	for i in 0..<len(canvas.cells) {
		canvas.cells[i] = false
	}
}

paint_canvas_at_mouse :: proc(canvas: ^Canvas, erase: bool, brush_size: int) -> bool {
	if !rl.IsMouseButtonDown(.LEFT) {
		return false
	}

	mouse_x := int(rl.GetMouseX())
	mouse_y := int(rl.GetMouseY())
	if mouse_x < CANVAS_X || mouse_y < CANVAS_Y {
		return false
	}

	col := (mouse_x - CANVAS_X) / PIXEL_SIZE
	row := (mouse_y - CANVAS_Y) / PIXEL_SIZE
	if col < 0 || col >= CANVAS_COLS || row < 0 || row >= CANVAS_ROWS {
		return false
	}

	next := !erase
	changed := false
	radius := brush_size / 2
	for brush_row in 0..<brush_size {
		for brush_col in 0..<brush_size {
			paint_col := col + brush_col - radius
			paint_row := row + brush_row - radius
			if paint_col < 0 || paint_col >= CANVAS_COLS || paint_row < 0 || paint_row >= CANVAS_ROWS {
				continue
			}

			index := canvas_index(paint_col, paint_row)
			if canvas.cells[index] != next {
				changed = true
			}
			canvas.cells[index] = next
		}
	}
	return changed
}

draw_canvas :: proc(canvas: ^Canvas, show_grid: bool) {
	canvas_w := CANVAS_COLS * PIXEL_SIZE
	canvas_h := CANVAS_ROWS * PIXEL_SIZE
	rl.DrawRectangle(CANVAS_X, CANVAS_Y, c.int(canvas_w), c.int(canvas_h), rl.Color{18, 18, 20, 255})

	for row in 0..<CANVAS_ROWS {
		for col in 0..<CANVAS_COLS {
			if canvas.cells[canvas_index(col, row)] {
				rl.DrawRectangle(
					c.int(CANVAS_X + col * PIXEL_SIZE),
					c.int(CANVAS_Y + row * PIXEL_SIZE),
					c.int(PIXEL_SIZE),
					c.int(PIXEL_SIZE),
					rl.Color{232, 224, 184, 255},
				)
			}
		}
	}

	if show_grid {
		for col in 0..=CANVAS_COLS {
			x := CANVAS_X + col * PIXEL_SIZE
			rl.DrawLine(c.int(x), CANVAS_Y, c.int(x), c.int(CANVAS_Y + canvas_h), rl.Color{42, 42, 46, 255})
		}
		for row in 0..=CANVAS_ROWS {
			y := CANVAS_Y + row * PIXEL_SIZE
			rl.DrawLine(CANVAS_X, c.int(y), c.int(CANVAS_X + canvas_w), c.int(y), rl.Color{42, 42, 46, 255})
		}
	}

	rl.DrawRectangleLines(CANVAS_X, CANVAS_Y, c.int(canvas_w), c.int(canvas_h), rl.Color{118, 118, 124, 255})
}

handle_controls :: proc(interaction: ^ui.UI_State, app: ^App_State, canvas: ^Canvas) {
	if ui.ui_activate(interaction, ID_CLEAR, ui.example_drawing_clear_rect()) {
		clear_canvas(canvas)
	}
	if ui.ui_activate(interaction, ID_ERASE, ui.example_drawing_erase_rect()) {
		app.erase = !app.erase
	}
	if ui.ui_activate(interaction, ID_GRID, ui.example_drawing_grid_rect()) {
		app.show_grid = !app.show_grid
	}
	if ui.ui_activate(interaction, ID_BRUSH_DOWN, ui.example_drawing_brush_down_rect()) {
		app.brush_size = ui.int_clamp(app.brush_size - 1, 1, MAX_BRUSH_SIZE)
	}
	if ui.ui_activate(interaction, ID_BRUSH_UP, ui.example_drawing_brush_up_rect()) {
		app.brush_size = ui.int_clamp(app.brush_size + 1, 1, MAX_BRUSH_SIZE)
	}
}

build_ui_commands :: proc(interaction: ^ui.UI_State, app: ^App_State, canvas: ^Canvas, command_storage: []ui.Draw_Command, warning_storage: []ui.Debug_Warning) -> []ui.Draw_Command {
	debug := ui.Debug_Buffer{warnings = warning_storage}
	draw := ui.draw_buffer(command_storage)
	frame := ui.ui_context(&draw, &debug)

	ui.example_drawing_controls_surface(
		&frame,
		app.erase,
		app.show_grid,
		app.brush_size,
		MAX_BRUSH_SIZE,
		clear_pressed = ui.ui_pressed(interaction, ID_CLEAR),
		erase_pressed = ui.ui_pressed(interaction, ID_ERASE),
		grid_pressed = ui.ui_pressed(interaction, ID_GRID),
		brush_down_pressed = ui.ui_pressed(interaction, ID_BRUSH_DOWN),
		brush_up_pressed = ui.ui_pressed(interaction, ID_BRUSH_UP),
	)

	return ui.ui_end(&draw)
}

main :: proc() {
	rl.InitWindow(WINDOW_WIDTH, WINDOW_HEIGHT, "8x16 drawing example")
	defer rl.CloseWindow()
	rl.SetTargetFPS(60)

	text_runes: [1024]rune
	renderer := rui.load_renderer(FONT_PATH, text_runes[:])
	defer rui.unload_renderer(&renderer)

	interaction: ui.UI_State
	app := App_State{show_grid = true, brush_size = 2}
	canvas: Canvas
	command_storage: [64]ui.Draw_Command
	warning_storage: [16]ui.Debug_Warning

	for !rl.WindowShouldClose() {
		ui.ui_begin_frame(&interaction, raylib_mouse_input())
		handle_controls(&interaction, &app, &canvas)
		if !interaction.hit_any {
			paint_canvas_at_mouse(&canvas, app.erase, app.brush_size)
		}

		commands := build_ui_commands(&interaction, &app, &canvas, command_storage[:], warning_storage[:])

		rl.BeginDrawing()
		rl.ClearBackground(rl.Color{12, 12, 14, 255})
		draw_canvas(&canvas, app.show_grid)
		rui.render_commands(&renderer, commands)
		rl.EndDrawing()
	}
}
