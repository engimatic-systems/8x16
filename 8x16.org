* Package =ui8x16=

The package boundary is intentionally small in PR 1: public value types only,
with no drawing backend dependency.

** Input State

Mouse input is ingested once per frame. The caller owns =UI_State= storage and
passes sampled mouse state into =ui_begin_frame=.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name mouse-button
Mouse_Button :: enum {
	Left,
	Middle,
	Right,
}
#+end_src

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name mouse-button-count
MOUSE_BUTTON_COUNT :: 3
#+end_src

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name mouse-input
Mouse_Input :: struct {
	x, y: int,
	down: [MOUSE_BUTTON_COUNT]bool,
}
#+end_src

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name ui-state
UI_State :: struct {
	mouse:         Mouse_Input,
	previous_down: [MOUSE_BUTTON_COUNT]bool,

	pressed_id: [MOUSE_BUTTON_COUNT]UI_ID,

	hit_any:       bool,
	activated_any: bool,
}
#+end_src

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name mouse-button-index
mouse_button_index :: proc(button: Mouse_Button) -> int {
	return int(button)
}
#+end_src

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name ui-begin-frame
ui_begin_frame :: proc(state: ^UI_State, input: Mouse_Input) {
	for i in 0..<MOUSE_BUTTON_COUNT {
		if state.previous_down[i] && !state.mouse.down[i] {
			state.pressed_id[i] = UI_ID_NONE
		}

		state.previous_down[i] = state.mouse.down[i]
	}

	state.mouse = input
	state.hit_any = false
	state.activated_any = false
}
#+end_src

** Activation

=ui_hit= marks advisory =hit_any=. =ui_activate= records press origin and
returns true only when press and release both occur inside the same =id, rect=
pair.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name ui-hit
ui_hit :: proc(state: ^UI_State, rect: Cell_Rect) -> bool {
	hit := contains(rect, state.mouse.x, state.mouse.y)
	if hit {
		state.hit_any = true
	}
	return hit
}
#+end_src

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name ui-activate
ui_activate :: proc(state: ^UI_State, id: UI_ID, rect: Cell_Rect, button: Mouse_Button = .Left) -> bool {
	index := mouse_button_index(button)
	hit := ui_hit(state, rect)
	pressed := !state.previous_down[index] && state.mouse.down[index]
	released := state.previous_down[index] && !state.mouse.down[index]

	if pressed && hit {
		state.pressed_id[index] = id
	}

	activated := released &&
	             hit &&
	             state.pressed_id[index] == id
	if activated {
		state.activated_any = true
	}
	return activated
}
#+end_src

** Geometry Constructors

Constructors assert positive dimensions. Callers should not pass invalid
rectangles; checked builds make that failure immediate.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name cell-rect-proc
cell_rect :: proc(col, row, w, h: int) -> Cell_Rect {
	assert(w > 0, "Cell_Rect width must be positive")
	assert(h > 0, "Cell_Rect height must be positive")
	return Cell_Rect{col = col, row = row, w = w, h = h}
}
#+end_src

=rect_rel= composes coordinates without retaining parent/child relationships.
The returned rectangle is absolute.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name rect-rel
rect_rel :: proc(parent: Cell_Rect, col, row, w, h: int) -> Cell_Rect {
	return cell_rect(parent.col + col, parent.row + row, w, h)
}
#+end_src

** Pixel Conversion

=rect_px= translates cell-space geometry to pixels using the fixed 8x16 metric.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name rect-px
rect_px :: proc(rect: Cell_Rect) -> Pixel_Rect {
	return Pixel_Rect{
		x = rect.col * CELL_WIDTH,
		y = rect.row * CELL_HEIGHT,
		w = rect.w * CELL_WIDTH,
		h = rect.h * CELL_HEIGHT,
	}
}
#+end_src

** Hit Testing

Hit testing is half-open in pixel space: left/top edges are inside, right/bottom
edges are outside.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name contains
contains :: proc(rect: Cell_Rect, x, y: int) -> bool {
	px := rect_px(rect)
	return x >= px.x &&
	       y >= px.y &&
	       x < px.x + px.w &&
	       y < px.y + px.h
}
#+end_src

** Primitive Draw Commands

The draw buffer is the handoff point between UI compilation and rendering. It
contains backend-ready primitive commands only: filled rectangles, stroked
rectangles, and text draws. Semantic UI concepts compile to these commands
before a renderer sees them.

Each primitive has its own payload type. This increases the public type surface,
but it names the renderable data directly and keeps each payload narrow.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name fill-rect-command
Fill_Rect_Command :: struct {
	rect:  Pixel_Rect,
	color: Color,
}
#+end_src

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name stroke-rect-command
Stroke_Rect_Command :: struct {
	rect:  Pixel_Rect,
	color: Color,
}
#+end_src

=Text_Command= carries a pixel origin rather than a rectangle. Text bounds and
overflow checks belong to layout/debug code before command emission; the backend
receives only the position, bytes, resolved color, and integer scale needed to
draw.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name text-command
Text_Command :: struct {
	pos:   Pixel_Point,
	text:  string,
	color: Color,
	scale: int,
}
#+end_src

=Draw_Command= is an Odin union over the primitive payloads. The tradeoff is
intentional: command storage may be larger and callers see more public types,
but the active-payload invariant is managed by Odin's type system instead of a
manual =kind= field plus inactive struct fields.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name draw-command
Draw_Command :: union {
	Fill_Rect_Command,
	Stroke_Rect_Command,
	Text_Command,
}
#+end_src

The buffer uses caller-owned storage. Commands are append-only and render in
buffer order; there is no z coordinate or hidden sorting step.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name draw-buffer
Draw_Buffer :: struct {
	commands: []Draw_Command,
	count:    int,
}
#+end_src

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name draw-buffer-proc
draw_buffer :: proc(storage: []Draw_Command) -> Draw_Buffer {
	return Draw_Buffer{commands = storage}
}
#+end_src

=text_scale_valid= constrains the fixed 8x16 design kernel to integer scales
=1|2|3|4=. Backends may choose how to realize those scales, but commands should
not carry arbitrary floating-point text sizes.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name text-scale-valid
text_scale_valid :: proc(scale: int) -> bool {
	return scale >= 1 && scale <= 4
}
#+end_src

=draw_push= is the only raw append path. Overflow asserts loudly because silent
command loss would make UI failures hard to diagnose.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name draw-push
draw_push :: proc(buffer: ^Draw_Buffer, command: Draw_Command) {
	assert(buffer.count < len(buffer.commands), "Draw_Buffer overflow")
	buffer.commands[buffer.count] = command
	buffer.count += 1
}
#+end_src

The =draw_*= procedures are typed constructors for command emission. They build
the correct union variant, append it, and keep call sites from spelling out the
union conversion directly.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name draw-fill-rect
draw_fill_rect :: proc(buffer: ^Draw_Buffer, rect: Pixel_Rect, color: Color) {
	draw_push(buffer, Draw_Command(Fill_Rect_Command{rect = rect, color = color}))
}
#+end_src

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name draw-stroke-rect
draw_stroke_rect :: proc(buffer: ^Draw_Buffer, rect: Pixel_Rect, color: Color) {
	draw_push(buffer, Draw_Command(Stroke_Rect_Command{rect = rect, color = color}))
}
#+end_src

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name draw-text
draw_text :: proc(buffer: ^Draw_Buffer, pos: Pixel_Point, text: string, color: Color, scale: int) {
	assert(text_scale_valid(scale), "Text scale must be 1..4")
	draw_push(buffer, Draw_Command(Text_Command{pos = pos, text = text, color = color, scale = scale}))
}
#+end_src

=ui_end= returns the ordered command slice for a renderer or test snapshot. It
does not clear storage; caller-owned frame lifetime remains explicit.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name ui-end
ui_end :: proc(buffer: ^Draw_Buffer) -> []Draw_Command {
	return buffer.commands[:buffer.count]
}
#+end_src
