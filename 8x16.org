** Input State

Mouse input is ingested once per frame. The caller owns =UI_State= storage and
passes sampled mouse state into =ui_begin_frame=.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name mouse-button
Mouse_Button :: enum {

	Middle,
	Right,
}
#+end_src

MOUSE_BUTTON_COUNT :: 3
#+end_src

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name mouse-input

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name ui-state
UI_State :: struct {
	mouse:         Mouse_Input,
	previous_down: [MOUSE_BUTTON_COUNT]bool,

	pressed_id: [MOUSE_BUTTON_COUNT]UI_ID,

	hit_any:       bool,
	activated_any: bool,
}
#+end_src

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name mouse-button-index
mouse_button_index :: proc(button: Mouse_Button) -> int {
	return int(button)
}
#+end_src

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name ui-begin-frame
ui_begin_frame :: proc(state: ^UI_State, input: Mouse_Input) {
	for i in 0..<MOUSE_BUTTON_COUNT {
		if state.previous_down[i] && !state.mouse.down[i] {
			state.pressed_id[i] = UI_ID_NONE
		}

		state.previous_down[i] = state.mouse.down[i]
	}

	state.mouse = input
	state.hit_any = false
	state.activated_any = false
}
#+end_src

** Activation

=ui_hit= marks advisory =hit_any=. =ui_activate= records press origin and
returns true only when press and release both occur inside the same =id, rect=
pair.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name ui-hit
ui_hit :: proc(state: ^UI_State, rect: Cell_Rect) -> bool {
	hit := contains(rect, state.mouse.x, state.mouse.y)
	if hit {
		state.hit_any = true
	}
	return hit
}
#+end_src

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name ui-activate
ui_activate :: proc(state: ^UI_State, id: UI_ID, rect: Cell_Rect, button: Mouse_Button = .Left) -> bool {
	index := mouse_button_index(button)
	hit := ui_hit(state, rect)
	pressed := !state.previous_down[index] && state.mouse.down[index]
	released := state.previous_down[index] && !state.mouse.down[index]

	if pressed && hit {
		state.pressed_id[index] = id
	}

	activated := released &&
	             hit &&
	             state.pressed_id[index] == id
	if activated {
		state.activated_any = true
	}
	return activated
}
#+end_src

** Geometry Constructors

Constructors assert positive dimensions. Callers should not pass invalid
rectangles; checked builds make that failure immediate.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name cell-rect-proc
cell_rect :: proc(col, row, w, h: int) -> Cell_Rect {
	assert(w > 0, "Cell_Rect width must be positive")
	assert(h > 0, "Cell_Rect height must be positive")
	return Cell_Rect{col = col, row = row, w = w, h = h}
}
#+end_src

=rect_rel= composes coordinates without retaining parent/child relationships.
The returned rectangle is absolute.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name rect-rel
rect_rel :: proc(parent: Cell_Rect, col, row, w, h: int) -> Cell_Rect {
	return cell_rect(parent.col + col, parent.row + row, w, h)
}
#+end_src

** Pixel Conversion

=rect_px= translates cell-space geometry to pixels using the fixed 8x16 metric.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name rect-px
rect_px :: proc(rect: Cell_Rect) -> Pixel_Rect {
	return Pixel_Rect{
		x = rect.col * CELL_WIDTH,
		y = rect.row * CELL_HEIGHT,
		w = rect.w * CELL_WIDTH,
		h = rect.h * CELL_HEIGHT,
	}
}
#+end_src

** Hit Testing

Hit testing is half-open in pixel space: left/top edges are inside, right/bottom
edges are outside.

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name contains
contains :: proc(rect: Cell_Rect, x, y: int) -> bool {
	px := rect_px(rect)
	return x >= px.x &&
	       y >= px.y &&
	       x < px.x + px.w &&
	       y < px.y + px.h
}
#+end_src

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name int-clamp
int_clamp :: proc(value, low, high: int) -> int {
	if value < low {
		return low
	}
	if value > high {
		return high
	}
	return value
}
#+end_src

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name debug-push
debug_push :: proc(buffer: ^Debug_Buffer, warning: Debug_Warning) -> bool {
	if buffer == nil {
		return false
	}
	if buffer.count >= len(buffer.warnings) {
		buffer.overflowed = true
		return false
	}

	buffer.warnings[buffer.count] = warning
	buffer.count += 1
	return true
}
#+end_src

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name draw-push
draw_push :: proc(buffer: ^Draw_Buffer, command: Draw_Command) -> bool {
	if buffer.count >= len(buffer.commands) {
		buffer.overflow_count += 1
		debug_push(buffer.debug, Debug_Warning{kind = .Draw_Command_Overflow})
		return false
	}

	buffer.commands[buffer.count] = command
	buffer.count += 1
	return true
}
#+end_src

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name draw-box
draw_box :: proc(buffer: ^Draw_Buffer, rect: Cell_Rect, fill, line: Color) -> bool {
	return draw_push(buffer, Draw_Command{
		kind = .Box,
		box = Draw_Box_Command{rect = rect_px(rect), fill = fill, line = line},
	})
}
#+end_src

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name draw-text
draw_text :: proc(buffer: ^Draw_Buffer, rect: Cell_Rect, text: string, color: Color) -> bool {
	return draw_push(buffer, Draw_Command{
		kind = .Text,
		text = Draw_Text_Command{rect = rect_px(rect), text = text, color = color},
	})
}
#+end_src

#+begin_src odin :tangle ui8x16/core.odin :noweb no :comments no :name draw-bar
draw_bar :: proc(buffer: ^Draw_Buffer, rect: Cell_Rect, fill_width_px: int, color, empty, line: Color) -> bool {
	px := rect_px(rect)
	fill_px := px
	fill_px.w = int_clamp(fill_width_px, 0, px.w)

	return draw_push(buffer, Draw_Command{
		kind = .Bar,
		bar = Draw_Bar_Command{rect = px, fill = fill_px, empty = empty, color = color, line = line},
	})
}
